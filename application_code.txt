# ---------------------------- Helper function for Gumbel estimation ----------------------------
fit_gumbel_pair <- function(R1, R2, pair_name) {
  # Input: two return vectors of same length
  # Output: list containing estimation results and marginal quantities
  
  n <- length(R1)
  
  # Thresholds at 90% quantile
  r1 <- as.numeric(quantile(R1, 0.9))
  r2 <- as.numeric(quantile(R2, 0.9))
  # Excesses over threshold
  R1_exc <- I(R1 > r1) * (R1 - r1)
  R2_exc <- I(R2 > r2) * (R2 - r2)
  
  # ---------------------------- Marginal model functions ----------------------------
  getsigma <- function(para, R) {
    n <- length(R)
    a <- para[1]; b <- para[2]; phi0 <- para[3]; phi1 <- para[4]; phi2 <- para[5]; phi3 <- para[6]
    logsigma <- rep(0, n)
    logsigma[1] <- (phi0 + phi2 / 2) / (1 - phi1)
    for (i in 1:(n - 1))
      logsigma[i + 1] <- phi0 + phi1 * logsigma[i] + phi2 * exp(-phi3 * R[i])
    return(exp(logsigma))
  }
  
  logik1 <- function(theta, R, ub, lb, r_thresh) {
    a <- theta[1]; b <- theta[2]
    n <- length(R)
    penal <- 1e10
    penalty <- penal * (sum(theta < lb) + sum(theta > ub))
    sigma <- getsigma(theta, R)
    logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r_thresh)^(b * sigma)) +
                       I(R > 0) * (1 / (1 + a * r_thresh)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
    return(-logik + penalty)
  }
  
  logik2 <- function(theta, R, ub, lb, r_thresh) {
    a <- theta[1]; b <- theta[2]
    n <- length(R)
    penal <- 1e10
    penalty <- penal * (sum(theta < lb) + sum(theta > ub))
    sigma <- getsigma(theta, R)
    logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r_thresh)^(b * sigma)) +
                       I(R > 0) * (1 / (1 + a * r_thresh)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
    return(-logik + penalty)
  }
  
  # Parameter bounds
  lb <- c(0, 0, -10, 0.0000001, 0.000000001, 0.0001)
  ub <- c(1000, 1000, 10, 1, 1, 100000)
  B <- 50
  
  # ---- First series ----
  finale1 <- foreach(i = 1:B, .combine = rbind) %dopar% {
    set.seed(i)
    loc <- runif(1)
    start <- loc * lb + (1 - loc) * ub
    fit <- nlminb(start, logik1, lower = lb, upper = ub,
                  control = list(eval.max = 1e7, iter.max = 1e7),
                  R = R1_exc, ub = ub, lb = lb, r_thresh = r1)
    c(fit$par, logik1(fit$par, R1_exc, ub, lb, r_thresh = r1))
  }
  para1 <- finale1[which.min(finale1[, 7]), 1:6]
  sigma1 <- getsigma(para1, R1_exc)
  xi1 <- sigma1 * para1[2]
  
  # ---- Second series ----
  finale2 <- foreach(i = 1:B, .combine = rbind) %dopar% {
    set.seed(i)
    loc <- runif(1)
    start <- loc * lb + (1 - loc) * ub
    fit <- nlminb(start, logik2, lower = lb, upper = ub,
                  control = list(eval.max = 1e7, iter.max = 1e7),
                  R = R2_exc, ub = ub, lb = lb, r_thresh = r2)
    c(fit$par, logik2(fit$par, R2_exc, ub, lb, r_thresh = r2))
  }
  para2 <- finale2[which.min(finale2[, 7]), 1:6]
  sigma2 <- getsigma(para2, R2_exc)
  xi2 <- sigma2 * para2[2]
  
  # ---------------------------- GPD functions ----------------------------
  dGPD <- function(y, tau, a, sigma, xi) {
    indicator <- y > 0
    (indicator == 0) * (1 - (1 / (1 + a * tau))^xi) +
      (indicator > 0) * ((1 / (1 + a * tau))^xi) * (xi / sigma) * (1 + y / sigma)^(-xi - 1)
  }
  pGPD <- function(y, tau, a, sigma, xi) {
    indicator <- y > 0
    (indicator == 0) * (1 - (1 / (1 + a * tau))^xi) +
      (indicator > 0) * (1 - ((1 / (1 + a * tau))^xi) * ((1 + y / sigma)^(-xi)))
  }
  
  # ---------------------------- Gumbel copula functions ----------------------------
  C_gumbel <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    exp(-(z1^alpha + z2^alpha)^(1/alpha))
  }
  dCdu_gumbel <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    C_val <- C_gumbel(u, v, alpha)
    term <- (z1^alpha + z2^alpha)^(1/alpha - 1) * z1^(alpha - 1)
    C_val * term / u
  }
  dCdv_gumbel <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    C_val <- C_gumbel(u, v, alpha)
    term <- (z1^alpha + z2^alpha)^(1/alpha - 1) * z2^(alpha - 1)
    C_val * term / v
  }
  d2C_gumbel <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    C_val <- C_gumbel(u, v, alpha)
    A <- (z1^alpha + z2^alpha)^(2/alpha - 2) * z1^(alpha-1) * z2^(alpha-1)
    B <- (alpha - 1) * (z1^alpha + z2^alpha)^(1/alpha - 2) * z1^(alpha-1) * z2^(alpha-1)
    C_val * (A + B) / (u * v)
  }
  
  # ---------------------------- GAS dynamics for alpha ----------------------------
  logistic <- function(alpha) log(alpha - 1)
  sigmoid <- function(lambda) 1 + exp(lambda)
  
  lPartialAlpha <- function(z1, z2, alpha) {
    (z1^alpha + z2^alpha)^(1/alpha) * (-1/alpha^2 * log(z1^alpha + z2^alpha) +
                                         1/alpha * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha))
  }
  logisticTailDepPartialAlpha <- function(z1, z2, alpha) -lPartialAlpha(z1, z2, alpha)
  logisticTailDepPartialAlpha1 <- function(z1, z2, alpha) {
    -lPartialAlpha(z1, z2, alpha) - 1/alpha^2 * log(z1^alpha + z2^alpha) +
      (1/alpha - 1) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha) + log(z1)
  }
  logisticTailDepPartialAlpha2 <- function(z1, z2, alpha) {
    -lPartialAlpha(z1, z2, alpha) - 1/alpha^2 * log(z1^alpha + z2^alpha) +
      (1/alpha - 1) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha) + log(z2)
  }
  logisticTailDepPartialAlpha12 <- function(z1, z2, alpha) {
    part1 <- -lPartialAlpha(z1, z2, alpha) + (log(z1) + log(z2)) - 1/alpha^2 * log(z1^alpha + z2^alpha)
    part2 <- (1/alpha - 2) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha)
    part3 <- (lPartialAlpha(z1, z2, alpha) + 1) / ((z1^alpha + z2^alpha)^(1/alpha) + alpha - 1)
    part1 + part2 + part3
  }
  
  recoverAlpha <- function(jointpara, yTrans, yRaw, init_alpha) {
    lower_bound <- 1 + 1e-5; upper_bound <- 10
    totalT <- ncol(yTrans)
    alpha <- numeric(totalT)
    beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
    alpha[1] <- ifelse(is.null(init_alpha), 2, init_alpha)
    for (t in 2:totalT) {
      z1 <- yTrans[1, t-1]; z2 <- yTrans[2, t-1]
      ind1 <- yRaw[1, t-1] > 0; ind2 <- yRaw[2, t-1] > 0
      if (!ind1 & !ind2) {
        innov <- logisticTailDepPartialAlpha(z1, z2, alpha[t-1]) * (alpha[t-1] - 1)
      } else if (ind1 & !ind2) {
        innov <- logisticTailDepPartialAlpha1(z1, z2, alpha[t-1]) * (alpha[t-1] - 1)
      } else if (!ind1 & ind2) {
        innov <- logisticTailDepPartialAlpha2(z1, z2, alpha[t-1]) * (alpha[t-1] - 1)
      } else {
        innov <- logisticTailDepPartialAlpha12(z1, z2, alpha[t-1]) * (alpha[t-1] - 1)
      }
      alpha[t] <- sigmoid(beta0 + beta1 * logistic(alpha[t-1]) + beta2 * innov)
      if (alpha[t] < lower_bound | is.na(alpha[t])) alpha[t] <- lower_bound
      if (alpha[t] > upper_bound | is.na(alpha[t])) alpha[t] <- upper_bound
    }
    alpha
  }
  
  # ---------------------------- Joint density ----------------------------
  dMPoT <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha) {
    u <- Fy1; v <- Fy2
    dens <- numeric(length(y1))
    idx00 <- y1==0 & y2==0
    idx10 <- y1>0 & y2==0
    idx01 <- y1==0 & y2>0
    idx11 <- y1>0 & y2>0
    if(any(idx00)) dens[idx00] <- C_gumbel(u[idx00], v[idx00], alpha[idx00])
    if(any(idx10)) dens[idx10] <- dCdu_gumbel(u[idx10], v[idx10], alpha[idx10]) * fy1[idx10]
    if(any(idx01)) dens[idx01] <- dCdv_gumbel(u[idx01], v[idx01], alpha[idx01]) * fy2[idx01]
    if(any(idx11)) dens[idx11] <- d2C_gumbel(u[idx11], v[idx11], alpha[idx11]) * fy1[idx11] * fy2[idx11]
    pmax(dens, 1e-300)
  }
  
  loglikBivariate <- function(jointpara, y, Fy, fy1, fy2, init_alpha) {
    alpha <- recoverAlpha(jointpara, yTrans = -log(Fy), yRaw = y, init_alpha = init_alpha)
    dens <- dMPoT(y[1,], y[2,], Fy[1,], Fy[2,], fy1, fy2, alpha)
    -sum(log(dens))
  }
  
  # ---------------------------- Initial static joint estimation ----------------------------
  init_upper_joint <- c(10, 1, 10, 1, 10)
  init_lower_joint <- c(0.1, 0.001, 0.1, 0.001, 1.001)
  tau_global <- c(r1, r2)
  observation <- rbind(R1_exc, R2_exc)
  
  initialEst_Joint <- function(para, a1, a2, y, tau) {
    tau1 <- tau[1]; xi1 <- para[1]; sigma1 <- para[2]
    tau2 <- tau[2]; xi2 <- para[3]; sigma2 <- para[4]
    alpha <- para[5]
    y1 <- y[1,]; y2 <- y[2,]
    Fy1 <- pGPD(y1, tau1, a1, sigma1, xi1)
    Fy2 <- pGPD(y2, tau2, a2, sigma2, xi2)
    fy1 <- dGPD(y1, tau1, a1, sigma1, xi1)
    fy2 <- dGPD(y2, tau2, a2, sigma2, xi2)
    dens <- dMPoT(y1, y2, Fy1, Fy2, fy1, fy2, rep(alpha, length(y1)))
    -sum(log(pmax(dens, 1e-300)))
  }
  
  temp0 <- c()
  for (rep in 1:10) {
    start <- init_lower_joint + (init_upper_joint - init_lower_joint) * runif(length(init_lower_joint))
    fit <- nlminb(start, initialEst_Joint, upper = init_upper_joint, lower = init_lower_joint,
                  tau = tau_global, y = observation, a1 = para1[1], a2 = para2[1])
    temp0 <- rbind(temp0, c(fit$par, fit$objective))
  }
  init_joint <- temp0[which.min(temp0[,6]), 1:5]
  
  # ---------------------------- Dynamic joint estimation ----------------------------
  Fy1 <- pGPD(R1_exc, r1, para1[1], sigma1, xi1)
  Fy2 <- pGPD(R2_exc, r2, para2[1], sigma2, xi2)
  fy1 <- dGPD(R1_exc, r1, para1[1], sigma1, xi1)
  fy2 <- dGPD(R2_exc, r2, para2[1], sigma2, xi2)
  
  Fy1 <- pGPD(R1_exc, r1, a1, sigma1, xi1)
  Fy2 <- pGPD(R2_exc, r2, a2, sigma2, xi2)
  fy1 <- dGPD(R1_exc, r1, a1, sigma1, xi1)
  fy2 <- dGPD(R2_exc, r2, a2, sigma2, xi2)
  Fy_est <- rbind(Fy1, Fy2)
  
  joint_upper <- c(0.1, 0.9999, 0.5)
  joint_lower <- c(-0.1, 0.9, -0.1)
  best_obj <- Inf; best_par <- NULL
  for (rep in 1:20) {
    start <- joint_lower + (joint_upper - joint_lower) * runif(3)
    fit <- nlminb(start, loglikBivariate,
                  upper = joint_upper, lower = joint_lower,
                  y = observation, Fy = Fy_est, fy1 = fy1, fy2 = fy2,
                  init_alpha = init_joint[5])
    if (fit$objective < best_obj) {
      best_obj <- fit$objective
      best_par <- fit$par
    }
  }
  est_joint <- best_par
  
  alpha_series <- recoverAlpha(est_joint, yTrans = -log(Fy_est), yRaw = observation, init_alpha = init_joint[5])
  
  # Compute summary statistics
  mean_alpha <- mean(alpha_series)
  var_alpha <- var(alpha_series)
  lambda <- 2 - 2^(1/alpha_series)
  mean_lambda <- mean(lambda)
  
  loglik_total <- -loglikBivariate(est_joint, y = observation, Fy = Fy_est,
                                   fy1 = fy1, fy2 = fy2, init_alpha = init_joint[5])
  k_total <- 15
  n_obs <- length(R1_exc)
  AIC_val <- -2 * loglik_total + 2 * k_total
  BIC_val <- -2 * loglik_total + k_total * log(n_obs)
  
  # Return all results
  list(
    pair_name = pair_name,
    marginal_series1 = list(para=para1, sigma=sigma1, xi=xi1, r=r1),
    marginal_series2 = list(para=para2, sigma=sigma2, xi=xi2, r=r2),
    Fy1 = Fy1, Fy2 = Fy2, fy1 = fy1, fy2 = fy2,
    joint_parameters = est_joint,
    alpha_series = alpha_series,
    mean_alpha = mean_alpha,
    var_alpha = var_alpha,
    mean_lambda = mean_lambda,
    logLik = loglik_total,
    AIC = AIC_val,
    BIC = BIC_val,
    n_obs = n_obs
  )
}

# ============================= MAIN LOOP =============================
out_summary <- "gumbel_results_summary.csv"
if(file.exists(out_summary)){
  done_pairs <- read.csv(out_summary)$pair_name
} else {
  done_pairs <- c()
}

all_res <- list()

for(pair in pairs){
  if(pair$name %in% done_pairs){
    cat("Skipping already processed pair:", pair$name, "\n")
    next
  }
  cat("\n========== Processing pair:", pair$name, "==========\n")
  
  tryCatch({
    R1 <- -data[, pair$col1]
    R2 <- -data[, pair$col2]
    
    res <- fit_gumbel_pair(R1, R2, pair$name)
    
    # Save marginal results for later use (Galambos/Clayton)
    marg_file <- paste0("marginal_", pair$name, ".RData")
    # Extract components to avoid save() issues with list elements
    marg1 <- res$marginal_series1
    marg2 <- res$marginal_series2
    Fy1 <- res$Fy1
    Fy2 <- res$Fy2
    fy1 <- res$fy1
    fy2 <- res$fy2
    save(marg1, marg2, Fy1, Fy2, fy1, fy2, file = marg_file)
    cat("Saved marginal results to", marg_file, "\n")
    
    # Append summary row
    all_res[[pair$name]] <- data.frame(
      pair_name = res$pair_name,
      mean_alpha = res$mean_alpha,
      var_alpha = res$var_alpha,
      mean_lambda = res$mean_lambda,
      beta0 = res$joint_parameters[1],
      beta1 = res$joint_parameters[2],
      beta2 = res$joint_parameters[3],
      logLik = res$logLik,
      AIC = res$AIC,
      BIC = res$BIC,
      n_obs = res$n_obs
    )
    
    current_df <- do.call(rbind, all_res)
    write.csv(current_df, out_summary, row.names = FALSE)
    
    cat("Finished pair:", pair$name, " | AIC =", round(res$AIC,2), "\n")
  }, error = function(e) {
    cat("ERROR for pair", pair$name, ":", e$message, "\n")
  })
}

cat("\nAll pairs processed. Final summary saved to", out_summary, "\n")

# ---------------------------- Helper: exceedances ----------------------------
get_exceedances <- function(ret_vec, r_thresh) {
  exc <- I(ret_vec > r_thresh) * (ret_vec - r_thresh)
  return(exc)
}

# ========================== 1. Galambos Copula ==========================
fit_galambos_from_marginal <- function(pair_name, col1, col2, data) {
  # Load marginal results saved from Gumbel estimation
  marg_file <- paste0("marginal_", pair_name, ".RData")
  if(!file.exists(marg_file)) stop("Marginal file not found: ", marg_file)
  load(marg_file)   # loads: marg1, marg2, Fy1, Fy2, fy1, fy2
  
  # Extract marginal parameters and thresholds
  r1 <- marg1$r; r2 <- marg2$r
  a1 <- marg1$para[1]; a2 <- marg2$para[1]
  sigma1 <- marg1$sigma; xi1 <- marg1$xi
  sigma2 <- marg2$sigma; xi2 <- marg2$xi
  
  # Exceedances
  R1_orig <- -data[, col1]
  R2_orig <- -data[, col2]
  R1_exc <- get_exceedances(R1_orig, r1)
  R2_exc <- get_exceedances(R2_orig, r2)
  observation <- rbind(R1_exc, R2_exc)
  
  # ---- Galambos copula functions ----
  calc_A <- function(z1, z2, alpha) {
    (z1^(-alpha) + z2^(-alpha))^(-1 / alpha)
  }
  pCopula_galambos <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    A <- calc_A(z1, z2, alpha)
    u * v * exp(A)
  }
  dCdu_galambos <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    A <- calc_A(z1, z2, alpha)
    B <- 1 - A^(1 + alpha) * z1^(-alpha - 1)
    v * exp(A) * B
  }
  d2C_galambos <- function(u, v, alpha) {
    z1 <- -log(u); z2 <- -log(v)
    A <- calc_A(z1, z2, alpha)
    term1 <- 1 - A^(1 + alpha) * z1^(-alpha - 1)
    term2 <- 1 - A^(1 + alpha) * z2^(-alpha - 1)
    exp(A) * (term1 * term2 + (1 + alpha) * A^(1 + 2 * alpha) * z1^(-alpha - 1) * z2^(-alpha - 1))
  }
  dMPoT_galambos <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha) {
    indicator1 <- (y1 == 0) & (y2 == 0)
    indicator2 <- (y1 > 0) & (y2 == 0)
    indicator3 <- (y1 == 0) & (y2 > 0)
    indicator4 <- (y1 > 0) & (y2 > 0)
    dens <- numeric(length(y1))
    if(any(indicator1)) dens[indicator1] <- pCopula_galambos(Fy1[indicator1], Fy2[indicator1], alpha[indicator1])
    if(any(indicator2)) dens[indicator2] <- fy1[indicator2] * dCdu_galambos(Fy1[indicator2], Fy2[indicator2], alpha[indicator2])
    if(any(indicator3)) dens[indicator3] <- fy2[indicator3] * dCdu_galambos(Fy2[indicator3], Fy1[indicator3], alpha[indicator3])
    if(any(indicator4)) dens[indicator4] <- fy1[indicator4] * fy2[indicator4] * d2C_galambos(Fy1[indicator4], Fy2[indicator4], alpha[indicator4])
    return(dens)
  }
  
  # Score via central difference
  score_alpha_galambos <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha, eps = 1e-5) {
    alpha_plus <- alpha + eps; alpha_minus <- alpha - eps
    f_plus <- dMPoT_galambos(y1, y2, Fy1, Fy2, fy1, fy2, alpha_plus)
    f_minus <- dMPoT_galambos(y1, y2, Fy1, Fy2, fy1, fy2, alpha_minus)
    f_plus <- pmax(f_plus, 1e-300); f_minus <- pmax(f_minus, 1e-300)
    (log(f_plus) - log(f_minus)) / (2 * eps)
  }
  
  # GAS recursion
  recoverAlpha_galambos <- function(jointpara, yRaw, Fy, fy1, fy2, init_alpha) {
    beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
    lower_bound <- 1e-5; upper_bound <- 20
    T <- ncol(yRaw)
    alpha <- numeric(T)
    alpha[1] <- init_alpha
    for(t in 2:T) {
      y1 <- yRaw[1, t-1]; y2 <- yRaw[2, t-1]
      Fy1t <- Fy[1, t-1]; Fy2t <- Fy[2, t-1]
      fy1t <- fy1[t-1]; fy2t <- fy2[t-1]
      s_alpha <- score_alpha_galambos(y1, y2, Fy1t, Fy2t, fy1t, fy2t, alpha[t-1])
      s_gamma <- s_alpha * alpha[t-1]
      gamma_prev <- log(alpha[t-1])
      gamma_new <- beta0 + beta1 * gamma_prev + beta2 * s_gamma
      alpha[t] <- exp(gamma_new)
      if(alpha[t] < lower_bound || is.na(alpha[t])) alpha[t] <- lower_bound
      if(alpha[t] > upper_bound || is.na(alpha[t])) alpha[t] <- upper_bound
    }
    alpha
  }
  
  loglikBivariate_galambos <- function(jointpara, y, Fy, fy1, fy2, init_alpha) {
    alpha <- recoverAlpha_galambos(jointpara, y, Fy, fy1, fy2, init_alpha)
    dens <- dMPoT_galambos(y[1,], y[2,], Fy[1,], Fy[2,], fy1, fy2, alpha)
    dens <- pmax(dens, 1e-300)
    -sum(log(dens))
  }
  
  # ---- Joint parameter estimation (GAS) ----
  joint_lower <- c(-2, 0.8, -1)
  joint_upper <- c(2, 0.9999, 1)
  reptimes <- 20
  init_alpha <- 2   # starting value for static initial guess, will be optimized
  
  # Use the marginal CDF/PDF already loaded (they were computed correctly)
  Fy_est <- rbind(Fy1, Fy2)
  fy1_est <- fy1
  fy2_est <- fy2
  
  best_obj <- Inf; best_par <- NULL
  for(rep in 1:reptimes) {
    start <- joint_lower + (joint_upper - joint_lower) * runif(3)
    fit <- nlminb(start, loglikBivariate_galambos,
                  upper = joint_upper, lower = joint_lower,
                  y = observation, Fy = Fy_est,
                  fy1 = fy1_est, fy2 = fy2_est,
                  init_alpha = init_alpha)
    if(fit$objective < best_obj) {
      best_obj <- fit$objective
      best_par <- fit$par
    }
  }
  est_joint <- best_par
  
  # Recover alpha series and statistics
  alpha_series <- recoverAlpha_galambos(est_joint, observation, Fy_est, fy1_est, fy2_est, init_alpha)
  mean_alpha <- mean(alpha_series)
  var_alpha <- var(alpha_series)
  lambda <- 2 - 2^(1/alpha_series)
  mean_lambda <- mean(lambda)
  
  loglik_total <- -loglikBivariate_galambos(est_joint, observation, Fy_est, fy1_est, fy2_est, init_alpha)
  k_total <- 15
  n_obs <- ncol(observation)
  AIC <- -2 * loglik_total + 2 * k_total
  BIC <- -2 * loglik_total + k_total * log(n_obs)
  
  list(pair_name = pair_name,
       joint_parameters = est_joint,
       alpha_series = alpha_series,
       mean_alpha = mean_alpha,
       var_alpha = var_alpha,
       mean_lambda = mean_lambda,
       logLik = loglik_total,
       AIC = AIC,
       BIC = BIC,
       n_obs = n_obs)
}

# ========================== 2. Survival Clayton Copula ==========================
fit_clayton_from_marginal <- function(pair_name, col1, col2, data) {
  marg_file <- paste0("marginal_", pair_name, ".RData")
  if(!file.exists(marg_file)) stop("Marginal file not found: ", marg_file)
  load(marg_file)   # marg1, marg2, Fy1, Fy2, fy1, fy2
  
  r1 <- marg1$r; r2 <- marg2$r
  a1 <- marg1$para[1]; a2 <- marg2$para[1]
  sigma1 <- marg1$sigma; xi1 <- marg1$xi
  sigma2 <- marg2$sigma; xi2 <- marg2$xi
  
  R1_orig <- -data[, col1]
  R2_orig <- -data[, col2]
  R1_exc <- get_exceedances(R1_orig, r1)
  R2_exc <- get_exceedances(R2_orig, r2)
  observation <- rbind(R1_exc, R2_exc)
  
  # ---- Survival Clayton functions ----
  get_w <- function(u, v, theta) {
    eps <- 1e-10
    s <- pmax(pmin(1 - u, 1 - eps), eps)
    t <- pmax(pmin(1 - v, 1 - eps), eps)
    s_inv <- s^(-theta); t_inv <- t^(-theta)
    A <- s_inv + t_inv - 1
    A <- pmax(A, eps)
    w <- A^(-1/theta)
    w <- pmin(w, 1/eps)
    list(w = w, A = A, s = s, t = t, s_inv = s_inv, t_inv = t_inv)
  }
  pSurvClayton <- function(u, v, theta) {
    w <- get_w(u, v, theta)$w
    u + v - 1 + w
  }
  dSurvClayton_du <- function(u, v, theta) {
    res <- get_w(u, v, theta)
    w <- res$w; s <- res$s
    term <- w^(theta + 1) * s^(-theta - 1)
    1 - term
  }
  dSurvClayton_dv <- function(u, v, theta) {
    res <- get_w(u, v, theta)
    w <- res$w; t <- res$t
    term <- w^(theta + 1) * t^(-theta - 1)
    1 - term
  }
  d2SurvClayton_dudv <- function(u, v, theta) {
    res <- get_w(u, v, theta)
    w <- res$w; s <- res$s; t <- res$t
    (theta + 1) * w^(2 * theta + 1) * s^(-theta - 1) * t^(-theta - 1)
  }
  # Joint density
  dMPoT_clayton <- function(y1, y2, Fy1, Fy2, fy1, fy2, theta) {
    indicator1 <- (y1 == 0) & (y2 == 0)
    indicator2 <- (y1 > 0) & (y2 == 0)
    indicator3 <- (y1 == 0) & (y2 > 0)
    indicator4 <- (y1 > 0) & (y2 > 0)
    dens <- numeric(length(y1))
    if(any(indicator1)) dens[indicator1] <- pSurvClayton(Fy1[indicator1], Fy2[indicator1], theta[indicator1])
    if(any(indicator2)) dens[indicator2] <- fy1[indicator2] * dSurvClayton_du(Fy1[indicator2], Fy2[indicator2], theta[indicator2])
    if(any(indicator3)) dens[indicator3] <- fy2[indicator3] * dSurvClayton_dv(Fy1[indicator3], Fy2[indicator3], theta[indicator3])
    if(any(indicator4)) dens[indicator4] <- fy1[indicator4] * fy2[indicator4] * d2SurvClayton_dudv(Fy1[indicator4], Fy2[indicator4], theta[indicator4])
    dens
  }
  
  # Score functions for GAS
  score_theta_clayton <- function(y1, y2, Fy1, Fy2, fy1, fy2, theta, eps = 1e-5) {
    theta_plus <- theta + eps; theta_minus <- theta - eps
    f_plus <- dMPoT_clayton(y1, y2, Fy1, Fy2, fy1, fy2, theta_plus)
    f_minus <- dMPoT_clayton(y1, y2, Fy1, Fy2, fy1, fy2, theta_minus)
    f_plus <- pmax(f_plus, 1e-300); f_minus <- pmax(f_minus, 1e-300)
    (log(f_plus) - log(f_minus)) / (2 * eps)
  }
  
  # GAS recursion for theta
  recoverTheta_clayton <- function(jointpara, yRaw, Fy, fy1, fy2, init_theta = 2) {
    beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
    lower_bound <- 1e-5; upper_bound <- 50
    T <- ncol(yRaw)
    theta <- numeric(T)
    gamma <- numeric(T)
    theta[1] <- init_theta
    gamma[1] <- log(theta[1])
    for(t in 2:T) {
      y1 <- yRaw[1, t-1]; y2 <- yRaw[2, t-1]
      Fy1t <- Fy[1, t-1]; Fy2t <- Fy[2, t-1]
      fy1t <- fy1[t-1]; fy2t <- fy2[t-1]
      s_theta <- score_theta_clayton(y1, y2, Fy1t, Fy2t, fy1t, fy2t, theta[t-1])
      s_gamma <- s_theta * theta[t-1]   # ∂logf/∂γ = (∂logf/∂θ)*θ
      gamma_new <- beta0 + beta1 * gamma[t-1] + beta2 * s_gamma
      theta[t] <- exp(gamma_new)
      if(theta[t] < lower_bound || is.na(theta[t])) theta[t] <- lower_bound
      if(theta[t] > upper_bound || is.na(theta[t])) theta[t] <- upper_bound
      gamma[t] <- log(theta[t])
    }
    theta
  }
  
  loglikBivariate_clayton <- function(jointpara, y, Fy, fy1, fy2, init_theta) {
    theta <- recoverTheta_clayton(jointpara, y, Fy, fy1, fy2, init_theta)
    dens <- dMPoT_clayton(y[1,], y[2,], Fy[1,], Fy[2,], fy1, fy2, theta)
    dens <- pmax(dens, 1e-300)
    -sum(log(dens))
  }
  
  # ---- Estimation ----
  joint_lower <- c(-2, 0.8, -1)   # beta0, beta1, beta2
  joint_upper <- c(2, 0.9999, 1)
  reptimes <- 20
  init_theta <- 2   # initial value for static start (used as init_theta for GAS)
  
  Fy_est <- rbind(Fy1, Fy2)
  fy1_est <- fy1; fy2_est <- fy2
  
  best_obj <- Inf; best_par <- NULL
  for(rep in 1:reptimes) {
    start <- joint_lower + (joint_upper - joint_lower) * runif(3)
    fit <- nlminb(start, loglikBivariate_clayton,
                  upper = joint_upper, lower = joint_lower,
                  y = observation, Fy = Fy_est,
                  fy1 = fy1_est, fy2 = fy2_est,
                  init_theta = init_theta)
    if(fit$objective < best_obj) {
      best_obj <- fit$objective
      best_par <- fit$par
    }
  }
  est_joint <- best_par
  
  theta_series <- recoverTheta_clayton(est_joint, observation, Fy_est, fy1_est, fy2_est, init_theta)
  mean_theta <- mean(theta_series)
  var_theta <- var(theta_series)
  lambda <- 2^(-1/theta_series)   # upper tail dependence for Survival Clayton
  mean_lambda <- mean(lambda)
  
  loglik_total <- -loglikBivariate_clayton(est_joint, observation, Fy_est, fy1_est, fy2_est, init_theta)
  k_total <- 15
  n_obs <- ncol(observation)
  AIC <- -2 * loglik_total + 2 * k_total
  BIC <- -2 * loglik_total + k_total * log(n_obs)
  
  list(pair_name = pair_name,
       joint_parameters = est_joint,
       theta_series = theta_series,
       mean_theta = mean_theta,
       var_theta = var_theta,
       mean_lambda = mean_lambda,
       logLik = loglik_total,
       AIC = AIC,
       BIC = BIC,
       n_obs = n_obs)
}
