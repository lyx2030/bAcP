r2 <- as.numeric(quantile(R2, 0.9))

# ---------------------------- Marginal model functions ----------------------------
# Generate sigma_t sequence
getsigma <- function(para, R) {
  n <- length(R)
  a <- para[1]; b <- para[2]; phi0 <- para[3]; phi1 <- para[4]; phi2 <- para[5]; phi3 <- para[6]
  logsigma <- rep(0, n)
  logsigma[1] <- (phi0 + phi2 / 2) / (1 - phi1)
  for (i in 1:(n - 1))
    logsigma[i + 1] <- phi0 + phi1 * logsigma[i] + phi2 * exp(-phi3 * R[i])
  return(exp(logsigma))
}

# Negative log-likelihood for first series
logik1 <- function(theta, R, ub, lb) {
  a <- theta[1]; b <- theta[2]
  n <- length(R)
  penal <- 1e10
  penalty <- penal * (sum(theta < lb) + sum(theta > ub))
  sigma <- getsigma(theta, R)
  # Mixed likelihood: probability mass for y=0, density for y>0
  logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r1)^(b * sigma)) +
                     I(R > 0) * (1 / (1 + a * r1)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
  return(-logik + penalty)
}

# Negative log-likelihood for second series
logik2 <- function(theta, R, ub, lb) {
  a <- theta[1]; b <- theta[2]
  n <- length(R)
  penal <- 1e10
  penalty <- penal * (sum(theta < lb) + sum(theta > ub))
  sigma <- getsigma(theta, R)
  logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r2)^(b * sigma)) +
                     I(R > 0) * (1 / (1 + a * r2)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
  return(-logik + penalty)
}

# ---------------------------- GPD functions ----------------------------
# PDF of GPD
dGPD <- function(y, tau, a, sigma, xi) {
  indicator <- y > 0
  result <- (indicator == 0) * (1 - (1 / (1 + a * tau))^xi) +
            (indicator > 0) * ((1 / (1 + a * tau))^xi) * (xi / sigma) * (1 + y / sigma)^(-xi - 1)
  return(result)
}
# Quantile of GPD
qGPD <- function(q, tau, a, sigma, xi) {
  indicator <- q < 1 - (tau / sigma)^(-xi)
  (indicator > 0) * 0 + (indicator == 0) * (sigma * ((1 - q)^(-1 / xi) / (1 + a * tau) - 1))
} 
# CDF of GPD 
pGPD <- function(y, tau, a, sigma, xi) {
  indicator <- y > 0
  result <- (indicator == 0) * (1 - (1 / (1 + a * tau))^xi) +
            (indicator > 0) * (1 - ((1 / (1 + a * tau))^xi) * ((1 + y / sigma)^(-xi)))
  return(result)
}

# Auxiliary functions for GAS transformation
logistic <- function(alpha) {
  return(log(alpha - 1))
}
sigmoid <- function(lambda) {
  return(1 + exp(lambda))
}

# ------------ Logistic tail dependence functions (Gumbel copula) -------------
logisticTailDep <- function(z1, z2, alpha) {
  return(exp(-(z1^alpha + z2^alpha)^(1 / alpha)))
}
logisticTailDepPartialDiff1 <- function(z1, z2, alpha) {
  return(-logisticTailDep(z1, z2, alpha) * (z1^alpha + z2^alpha)^(1 / alpha - 1) * z1^(alpha - 1))
}
logisticTailDepPartialDiff2 <- function(z1, z2, alpha) {
  return(-logisticTailDep(z1, z2, alpha) * (z1^alpha + z2^alpha)^(1 / alpha - 1) * z2^(alpha - 1))
}
logisticTailDepPartialDiff12 <- function(z1, z2, alpha) {
  part1 <- logisticTailDep(z1, z2, alpha) * (z1^alpha + z2^alpha)^(2 / alpha - 2) * z2^(alpha - 1) * z1^(alpha - 1)
  part2 <- logisticTailDep(z1, z2, alpha) * (z1^alpha + z2^alpha)^(1 / alpha - 2) * z2^(alpha - 1) * z1^(alpha - 1) * (alpha - 1)
  return(part1 + part2)
}

# Partial derivatives of log PDF w.r.t. alpha (for GAS)
lPartialAlpha <- function(z1, z2, alpha) {
  return((z1^alpha + z2^alpha)^(1 / alpha) * (-1 / alpha^2 * log(z1^alpha + z2^alpha) +
          1 / alpha * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha)))
}
logisticTailDepPartialAlpha <- function(z1, z2, alpha) {
  return(-lPartialAlpha(z1, z2, alpha))
}
logisticTailDepPartialAlpha1 <- function(z1, z2, alpha) {
  return(-lPartialAlpha(z1, z2, alpha) - 1 / alpha^2 * log(z1^alpha + z2^alpha) + 
           (1 / alpha - 1) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha) + log(z1))
}
logisticTailDepPartialAlpha2 <- function(z1, z2, alpha) {
  return(-lPartialAlpha(z1, z2, alpha) - 1 / alpha^2 * log(z1^alpha + z2^alpha) + 
           (1 / alpha - 1) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha) + log(z2))
}
logisticTailDepPartialAlpha12 <- function(z1, z2, alpha) {
  part1 <- -lPartialAlpha(z1, z2, alpha) + (log(z1) + log(z2)) - 1 / alpha^2 * log(z1^alpha + z2^alpha)
  part2 <- (1 / alpha - 2) * (z1^alpha * log(z1) + z2^alpha * log(z2)) / (z1^alpha + z2^alpha)
  part3 <- (lPartialAlpha(z1, z2, alpha) + 1) / ((z1^alpha + z2^alpha)^(1 / alpha) + alpha - 1)
  return(part1 + part2 + part3)
}

# ------------------ Recover alpha series (GAS dynamics) ------------------------
recoverAlpha <- function(jointpara, yTrans, yRaw = NULL, init = NULL) {
  # Constrain alpha to [lower_bound, upper_bound] for numerical stability
  lower_bound <- 1 + 1e-5; upper_bound <- 10
  totalT <- dim(yTrans)[2]
  alpha <- rep(0, totalT)
  beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
  if (is.null(init)) {
    alpha[1] <- 2
  } else {
    alpha[1] <- init
  }
  for (t in 2:totalT) {
    z1 <- yTrans[1, (t - 1)]; z2 <- yTrans[2, (t - 1)]
    indicator1 <- yRaw[1, (t - 1)] > 0; indicator2 <- yRaw[2, (t - 1)] > 0
    if (!indicator1 & !indicator2) { # y1=0, y2=0
      innovation <- logisticTailDepPartialAlpha(z1, z2, alpha[t - 1]) * (alpha[t - 1] - 1)
    }
    if (indicator1 & !indicator2) {  # y1>0, y2=0
      innovation <- logisticTailDepPartialAlpha1(z1, z2, alpha[t - 1]) * (alpha[t - 1] - 1)
    }
    if (!indicator1 & indicator2) {  # y1=0, y2>0
      innovation <- logisticTailDepPartialAlpha2(z1, z2, alpha[t - 1]) * (alpha[t - 1] - 1)
    }
    if (indicator1 & indicator2) {   # y1>0, y2>0
      innovation <- logisticTailDepPartialAlpha12(z1, z2, alpha[t - 1]) * (alpha[t - 1] - 1)
    }
    alpha[t] <- sigmoid(beta0 + beta1 * logistic(alpha[t - 1]) + beta2 * innovation)
    if (alpha[t] < lower_bound | is.na(alpha[t])) { alpha[t] <- lower_bound }
    if (alpha[t] > upper_bound | is.na(alpha[t])) { alpha[t] <- upper_bound }
  }
  return(alpha)
}

# --------------- Joint density for different observation types --------------
dMPoT <- function(y1, y2, Fy1, Fy2, alpha) {
  indicator1 <- (y1 == 0) & (y2 == 0)
  indicator2 <- (y1 > 0) & (y2 == 0)
  indicator3 <- (y1 == 0) & (y2 > 0)
  indicator4 <- (y1 > 0) & (y2 > 0)
  z1 <- -log(Fy1)
  z2 <- -log(Fy2)
  # Likelihood contributions by case
  part1 <- logisticTailDep(z1 = z1[indicator1], z2 = z2[indicator1], alpha = alpha[indicator1])
  part2 <- -logisticTailDepPartialDiff1(z1 = z1[indicator2], z2 = z2[indicator2], alpha = alpha[indicator2])
  part3 <- -logisticTailDepPartialDiff2(z1 = z1[indicator3], z2 = z2[indicator3], alpha = alpha[indicator3])
  part4 <- logisticTailDepPartialDiff12(z1 = z1[indicator4], z2 = z2[indicator4], alpha = alpha[indicator4])
  return(c(part1, part2, part3, part4))
}

# Negative log-likelihood for joint parameters (second stage)
loglikBivariate <- function(jointpara, y, Fy, yTrans, init = NULL) {
  alpha <- recoverAlpha(jointpara = jointpara, yTrans = yTrans, yRaw = y, init)
  loglik <- -sum(log(dMPoT(y1 = y[1, ], y2 = y[2, ], Fy1 = Fy[1, ],
                           Fy2 = Fy[2, ], alpha = alpha)))
  return(loglik)
}

# ---------------- Initial static joint estimation ------------------
initialEst_Joint <- function(para, a1, a2, y, tau) {
  tau1 <- tau[1]; xi1 <- para[1]; sigma1 <- para[2]
  tau2 <- tau[2]; xi2 <- para[3]; sigma2 <- para[4]
  alpha <- para[5]
  y1 <- y[1, ]; y2 <- y[2, ]
  Fy1 <- pGPD(y1, tau1, a1, sigma1, xi1)
  Fy2 <- pGPD(y2, tau2, a2, sigma2, xi2)
  fy1 <- dGPD(y1, tau1, a1, sigma1, xi1)
  fy2 <- dGPD(y2, tau2, a2, sigma2, xi2)
  loglik <- -sum(log(dMPoTInit(y1 = y1, y2 = y2, Fy1 = Fy1, Fy2 = Fy2,
                                fy1 = fy1, fy2 = fy2, alpha = alpha)))
  return(loglik)
}

dMPoTInit <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha) {
  indicator1 <- (y1 == 0) & (y2 == 0)
  indicator2 <- (y1 > 0) & (y2 == 0)
  indicator3 <- (y1 == 0) & (y2 > 0)
  indicator4 <- (y1 > 0) & (y2 > 0)
  z1 <- -log(Fy1)
  z2 <- -log(Fy2)
  # Likelihood contributions by case
  part1 <- logisticTailDep(z1 = z1[indicator1], z2 = z2[indicator1], alpha = alpha)
  part2 <- -logisticTailDepPartialDiff1(z1 = z1[indicator2], z2 = z2[indicator2], alpha = alpha) * (fy1[indicator2] / Fy1[indicator2])
  part3 <- -logisticTailDepPartialDiff2(z1 = z1[indicator3], z2 = z2[indicator3], alpha = alpha) * (fy2[indicator3] / Fy2[indicator3])
  part4 <- logisticTailDepPartialDiff12(z1 = z1[indicator4], z2 = z2[indicator4], alpha = alpha) * (fy1[indicator4] / Fy1[indicator4]) * (fy2[indicator4] / Fy2[indicator4])
  return(c(part1, part2, part3, part4))
}

# ---------------------------- Galambos Copula functions ----------------------------
# Auxiliary quantity A = (z1^{-alpha} + z2^{-alpha})^{-1/alpha}
calc_A <- function(z1, z2, alpha) {
  (z1^(-alpha) + z2^(-alpha))^(-1 / alpha)
}

# Galambos copula distribution function C(u,v)
pCopula_galambos <- function(u, v, alpha) {
  z1 <- -log(u)
  z2 <- -log(v)
  A <- calc_A(z1, z2, alpha)
  u * v * exp(A)
}

# First-order partial derivative dC/du
dCdu_galambos <- function(u, v, alpha) {
  z1 <- -log(u)
  z2 <- -log(v)
  A <- calc_A(z1, z2, alpha)
  B <- 1 - A^(1 + alpha) * z1^(-alpha - 1)
  v * exp(A) * B
}

# Second-order mixed partial derivative d^2C/dudv
d2C_galambos <- function(u, v, alpha) {
  z1 <- -log(u)
  z2 <- -log(v)
  A <- calc_A(z1, z2, alpha)
  term1 <- 1 - A^(1 + alpha) * z1^(-alpha - 1)
  term2 <- 1 - A^(1 + alpha) * z2^(-alpha - 1)
  exp(A) * (term1 * term2 + (1 + alpha) * A^(1 + 2 * alpha) * z1^(-alpha - 1) * z2^(-alpha - 1))
}

# Joint probability/density function (four cases)
dMPoT_galambos <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha) {
  indicator1 <- (y1 == 0) & (y2 == 0)
  indicator2 <- (y1 > 0) & (y2 == 0)
  indicator3 <- (y1 == 0) & (y2 > 0)
  indicator4 <- (y1 > 0) & (y2 > 0)
  
  n <- length(y1)
  dens <- numeric(n)
  
  if (any(indicator1)) {
    dens[indicator1] <- pCopula_galambos(Fy1[indicator1], Fy2[indicator1], alpha[indicator1])
  }
  if (any(indicator2)) {
    dens[indicator2] <- fy1[indicator2] * dCdu_galambos(Fy1[indicator2], Fy2[indicator2], alpha[indicator2])
  }
  if (any(indicator3)) {
    dens[indicator3] <- fy2[indicator3] * dCdu_galambos(Fy2[indicator3], Fy1[indicator3], alpha[indicator3])  # symmetry
  }
  if (any(indicator4)) {
    dens[indicator4] <- fy1[indicator4] * fy2[indicator4] * d2C_galambos(Fy1[indicator4], Fy2[indicator4], alpha[indicator4])
  }
  return(dens)
}

# Numerical differentiation for ∂log f/∂α (central difference)
score_alpha_galambos <- function(y1, y2, Fy1, Fy2, fy1, fy2, alpha, eps = 1e-5) {
  alpha_plus <- alpha + eps
  alpha_minus <- alpha - eps
  f_plus <- dMPoT_galambos(y1, y2, Fy1, Fy2, fy1, fy2, alpha_plus)
  f_minus <- dMPoT_galambos(y1, y2, Fy1, Fy2, fy1, fy2, alpha_minus)
  # Avoid log(0) due to zero values
  f_plus <- pmax(f_plus, 1e-300)
  f_minus <- pmax(f_minus, 1e-300)
  (log(f_plus) - log(f_minus)) / (2 * eps)
}

# --------------- GAS dynamic update for α_t (α = exp(γ)) -----------------
recoverAlpha_galambos <- function(jointpara, yRaw, Fy, fy1, fy2, init_alpha = 2) {
  # jointpara: (beta0, beta1, beta2)
  beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
  lower_bound <- 1e-5   # α > 0
  upper_bound <- 20     # prevent overflow
  totalT <- ncol(yRaw)
  alpha <- numeric(totalT)
  alpha[1] <- init_alpha
  
  for (t in 2:totalT) {
    # Use observation at time t-1 to compute the score
    y1 <- yRaw[1, t-1]; y2 <- yRaw[2, t-1]
    Fy1t <- Fy[1, t-1]; Fy2t <- Fy[2, t-1]
    fy1t <- fy1[t-1]; fy2t <- fy2[t-1]
    
    s_alpha <- score_alpha_galambos(y1, y2, Fy1t, Fy2t, fy1t, fy2t, alpha[t-1])
    s_gamma <- s_alpha * alpha[t-1]   # dγ = dα/α, actually ∂logf/∂γ = (∂logf/∂α) * (∂α/∂γ) = s_alpha * α
    
    gamma_prev <- log(alpha[t-1])
    gamma_new <- beta0 + beta1 * gamma_prev + beta2 * s_gamma
    alpha[t] <- exp(gamma_new)
    
    # Boundary constraints
    if (alpha[t] < lower_bound || is.na(alpha[t])) alpha[t] <- lower_bound
    if (alpha[t] > upper_bound || is.na(alpha[t])) alpha[t] <- upper_bound
  }
  return(alpha)
}

# Negative log-likelihood function (joint parameters)
loglikBivariate_galambos <- function(jointpara, y, Fy, fy1, fy2, init_alpha = 2) {
  alpha <- recoverAlpha_galambos(jointpara, y, Fy, fy1, fy2, init_alpha)
  dens <- dMPoT_galambos(y[1, ], y[2, ], Fy[1, ], Fy[2, ], fy1, fy2, alpha)
  dens <- pmax(dens, 1e-300)   # prevent -Inf
  loglik <- -sum(log(dens))
  return(loglik)
}

# ---------------------------- Marginal model functions ----------------------------
# Generate sigma_t series
getsigma <- function(para, R) {
  n <- length(R)
  a <- para[1]; b <- para[2]; phi0 <- para[3]; phi1 <- para[4]; phi2 <- para[5]; phi3 <- para[6]
  logsigma <- rep(0, n)
  logsigma[1] <- (phi0 + phi2 / 2) / (1 - phi1)
  for (i in 1:(n - 1))
    logsigma[i + 1] <- phi0 + phi1 * logsigma[i] + phi2 * exp(-phi3 * R[i])
  return(exp(logsigma))
}

# Negative log-likelihood function (marginal)
logik1 <- function(theta, R, ub, lb) {
  a <- theta[1]; b <- theta[2]
  n <- length(R)
  penal <- 1e10
  penalty <- penal * (sum(theta < lb) + sum(theta > ub))
  sigma <- getsigma(theta, R)
  # Mixed likelihood: y=0 gives probability mass, y>0 gives density
  logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r1)^(b * sigma)) +
                     I(R > 0) * (1 / (1 + a * r1)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
  return(-logik + penalty)
}

logik2 <- function(theta, R, ub, lb) {
  a <- theta[1]; b <- theta[2]
  n <- length(R)
  penal <- 1e10
  penalty <- penal * (sum(theta < lb) + sum(theta > ub))
  sigma <- getsigma(theta, R)
  logik <- sum(log(I(R == 0) * (1 - 1 / (1 + a * r2)^(b * sigma)) +
                     I(R > 0) * (1 / (1 + a * r2)^(b * sigma) * b * (1 + R / sigma)^(-b * sigma - 1)))) / n
  return(-logik + penalty)
}

# ----------------------- Marginal distribution functions ------------------------
# GPD mixture CDF
pGPD <- function(y, tau, a, sigma, xi) {
  indicator <- y > 0
  result <- (indicator == 0) * (1 - 1 / (1 + a * tau)^xi) +
    (indicator > 0) * (1 - (1 / (1 + a * tau)^xi) * (1 + y / sigma)^(-xi))
  return(result)
}

# GPD mixture PDF
dGPD <- function(y, tau, a, sigma, xi) {
  indicator <- y > 0
  result <- (indicator == 0) * (1 - 1 / (1 + a * tau)^xi) +
    (indicator > 0) * (1 / (1 + a * tau)^xi) * (xi / sigma) * (1 + y / sigma)^(-xi - 1)
  return(result)
}

# ------------- Basic Copula Functions -----------------------------
# Helper function: compute w = [(1-u)^{-theta} + (1-v)^{-theta} - 1]^{-1/theta}
get_w <- function(u, v, theta) {
  eps <- 1e-10
  s <- pmax(pmin(1 - u, 1 - eps), eps)   # ensure s in [eps, 1-eps]
  t <- pmax(pmin(1 - v, 1 - eps), eps)
  s_inv <- s^(-theta)
  t_inv <- t^(-theta)
  A <- s_inv + t_inv - 1
  A <- pmax(A, eps)                       # prevent A from being negative
  w <- A^(-1/theta)
  w <- pmin(w, 1/eps)                      # bound w from above
  list(w = w, A = A, s = s, t = t, s_inv = s_inv, t_inv = t_inv)
}

# Joint CDF: C(u,v)
pSurvClayton <- function(u, v, theta) {
  w <- get_w(u, v, theta)$w
  u + v - 1 + w
}

# Partial derivative dC/du
dSurvClayton_du <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; s <- res$s
  term <- w^(theta + 1) * s^(-theta - 1)
  1 - term
}

# Partial derivative dC/dv
dSurvClayton_dv <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; t <- res$t
  term <- w^(theta + 1) * t^(-theta - 1)
  1 - term
}

# Second-order mixed partial d2C/dudv (density)
d2SurvClayton_dudv <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; s <- res$s; t <- res$t
  (theta + 1) * w^(2 * theta + 1) * s^(-theta - 1) * t^(-theta - 1)
}

# ----------------- Derivatives w.r.t. theta (for score) --------------------
dw_dtheta <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; A <- res$A; s <- res$s; t <- res$t
  s_inv <- res$s_inv; t_inv <- res$t_inv
  log_s <- log(s); log_t <- log(t)
  numerator <- s_inv * log_s + t_inv * log_t
  term <- numerator / A
  (w / theta) * (term - log(w))
}

# d log C / d theta (case: y1=0, y2=0)
dlogC_dtheta <- function(u, v, theta) {
  C <- pSurvClayton(u, v, theta)
  if (C <= 0) C <- 1e-300
  dw <- dw_dtheta(u, v, theta)
  dw / C
}

# d log(dCdu) / d theta (case: y1>0, y2=0)
dlogdCdu_dtheta <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; s <- res$s
  log_w <- log(w); log_s <- log(s)
  B <- w^(theta + 1) * s^(-theta - 1)
  one_minus_B <- 1 - B
  if (one_minus_B < 1e-10) one_minus_B <- 1e-10
  dw <- dw_dtheta(u, v, theta)
  dB <- B * ((log_w - log_s) + (theta + 1) * dw / w)
  -dB / one_minus_B
}

# d log(dCdv) / d theta (case: y1=0, y2>0)
dlogdCdv_dtheta <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; t <- res$t
  log_w <- log(w); log_t <- log(t)
  B <- w^(theta + 1) * t^(-theta - 1)
  one_minus_B <- 1 - B
  if (one_minus_B < 1e-10) one_minus_B <- 1e-10
  dw <- dw_dtheta(u, v, theta)
  dB <- B * ((log_w - log_t) + (theta + 1) * dw / w)
  -dB / one_minus_B
}

# d log c / d theta (case: y1>0, y2>0)
dlogc_dtheta <- function(u, v, theta) {
  res <- get_w(u, v, theta)
  w <- res$w; s <- res$s; t <- res$t
  log_w <- log(w); log_s <- log(s); log_t <- log(t)
  dw <- dw_dtheta(u, v, theta)
  1 / (theta + 1) + 2 * log_w + (2 * theta + 1) * dw / w - (log_s + log_t)
}

# ----------------------------- GAS Dynamics for theta -----------------------------
recoverTheta <- function(jointpara, y, Fy, init = NULL) {
  # jointpara: c(beta0, beta1, beta2)
  # y: 2 x T matrix of raw observations
  # Fy: 2 x T matrix of marginal CDFs
  # init: initial theta[1] (default 2)
  lower_bound <- 1e-5
  upper_bound <- 100
  totalT <- ncol(y)
  theta <- numeric(totalT)
  gamma <- numeric(totalT)
  beta0 <- jointpara[1]; beta1 <- jointpara[2]; beta2 <- jointpara[3]
  
  if (is.null(init)) {
    theta[1] <- 2
  } else {
    theta[1] <- init
  }
  gamma[1] <- log(theta[1])
  
  for (t in 2:totalT) {
    u1 <- Fy[1, t-1]
    u2 <- Fy[2, t-1]
    y1 <- y[1, t-1]
    y2 <- y[2, t-1]
    theta_prev <- theta[t-1]
    
    # Compute score dlogL/dtheta at previous time
    if (y1 == 0 && y2 == 0) {
      dlogL_dtheta <- dlogC_dtheta(u1, u2, theta_prev)
    } else if (y1 > 0 && y2 == 0) {
      dlogL_dtheta <- dlogdCdu_dtheta(u1, u2, theta_prev)
    } else if (y1 == 0 && y2 > 0) {
      dlogL_dtheta <- dlogdCdv_dtheta(u1, u2, theta_prev)
    } else {
      dlogL_dtheta <- dlogc_dtheta(u1, u2, theta_prev)
    }
    # Score w.r.t. gamma: dlogL/dgamma = dlogL/dtheta * theta
    score_gamma <- dlogL_dtheta * theta_prev
    if (is.na(score_gamma) || is.infinite(score_gamma)) score_gamma <- 0
    
    # Update gamma
    gamma[t] <- beta0 + beta1 * gamma[t-1] + beta2 * score_gamma
    theta[t] <- exp(gamma[t])
    
    # Bound constraints
    if (theta[t] < lower_bound) theta[t] <- lower_bound
    if (theta[t] > upper_bound) theta[t] <- upper_bound
  }
  return(theta)
}

# --------------- Initial Constant theta Estimation --------------------
initialEst_JointClayton <- function(para, a1, a2, y, tau) {
  # para: c(xi1, sigma1, xi2, sigma2, theta)
  tau1 <- tau[1]; tau2 <- tau[2]
  xi1 <- para[1]; sigma1 <- para[2]
  xi2 <- para[3]; sigma2 <- para[4]
  theta <- para[5]
  y1 <- y[1, ]; y2 <- y[2, ]
  
  Fy1 <- pmin(pmax(pGPD(y1, tau1, a1, sigma1, xi1), 1e-10), 1 - 1e-10)
  Fy2 <- pmin(pmax(pGPD(y2, tau2, a2, sigma2, xi2), 1e-10), 1 - 1e-10)
  fy1 <- dGPD(y1, tau1, a1, sigma1, xi1)
  fy2 <- dGPD(y2, tau2, a2, sigma2, xi2)
  
  loglik <- 0
  for (t in 1:length(y1)) {
    u1 <- Fy1[t]; u2 <- Fy2[t]
    if (y1[t] == 0 && y2[t] == 0) {
      cop <- pSurvClayton(u1, u2, theta)
    } else if (y1[t] > 0 && y2[t] == 0) {
      cop <- dSurvClayton_du(u1, u2, theta) * fy1[t]
    } else if (y1[t] == 0 && y2[t] > 0) {
      cop <- dSurvClayton_dv(u1, u2, theta) * fy2[t]
    } else {
      cop <- d2SurvClayton_dudv(u1, u2, theta) * fy1[t] * fy2[t]
    }
    if (is.na(cop) || is.nan(cop) || cop <= 0) cop <- 1e-300
    loglik <- loglik + log(cop)
  }
  return(-loglik)
}
